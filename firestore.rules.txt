rules_version = '2';
// Black Tap Bar Book: Firestore security rules, version 2.
//
// Identity is the signed-in Firebase Auth uid. Membership is a staff/{uid} record with a role:
// owner (one account, claimed once through config/owner), manager, or staff. Removing that record removes
// all access. Names on records are display snapshots; the uid fields are what these rules check.
//
// Anything not matched below is denied, including every collection that isn't listed.
//
// Batched writes are kept to 20 document lookups or fewer even if Firestore caches nothing between the
// writes in a batch (the documented limit). The heaviest is a PIN reset at 15.
service cloud.firestore {
  match /databases/{database}/documents {

    // ---------- Who is asking ----------
    function signedIn() { return request.auth != null; }
    function staffDoc(uid) { return /databases/$(database)/documents/staff/$(uid); }
    function nameDoc(username) { return /databases/$(database)/documents/usernames/$(username); }
    function ownerDoc() { return /databases/$(database)/documents/config/owner; }
    function auditDoc(id) { return /databases/$(database)/documents/audit/$(id); }
    // The caller's staff record, or null when they aren't on the team.
    function profile() { return get(staffDoc(request.auth.uid)); }
    function isMember() { return signedIn() && profile() != null; }
    function isManager() { return isMember() && profile().data.role in ['manager', 'owner']; }
    function isOwner() { return isMember() && profile().data.role == 'owner'; }
    // True unless uid belongs to the owner. Someone who has left counts as not the owner.
    function notOwner(uid) { let p = get(staffDoc(uid)); return p == null || p.data.role != 'owner'; }

    // ---------- Field checks ----------
    function incoming() { return request.resource.data; }
    function exactly(keys) { return incoming().keys().hasAll(keys) && incoming().keys().hasOnly(keys); }
    function changesOnly(keys) { return incoming().diff(resource.data).affectedKeys().hasOnly(keys); }
    // Required text needs something besides spaces. Optional text may be empty.
    function reqText(s, max) { return s is string && s.trim().size() > 0 && s.size() <= max; }
    function optText(s, max) { return s is string && s.size() <= max; }
    // timestamp.date() refuses impossible dates such as 2026-02-31, so this also checks the date is real.
    function dayOf(s) { return timestamp.date(int(s[0:4]), int(s[5:7]), int(s[8:10])); }
    function isDay(s) { return s is string && s.matches('[0-9]{4}-[0-9]{2}-[0-9]{2}'); }
    function isTime(s) { return s is string && s.matches('([01][0-9]|2[0-3]):[0-5][0-9]'); }
    // Written by the caller: their uid, their name as it is on their staff record, and the server's clock.
    function stamped() {
      return incoming().uid == request.auth.uid && incoming().name == profile().data.name && incoming().at == request.time;
    }
    // The same for a change, under that change's own field names.
    function signedAs(nameKey, uidKey, atKey) {
      return incoming()[nameKey] == profile().data.name && incoming()[uidKey] == request.auth.uid && incoming()[atKey] == request.time;
    }
    // Lower case, with every run of other characters turned into one dash. The app does exactly the same.
    function slug(s) { return s.lower().replace('[^a-z0-9]+', '-'); }
    function autoId(s) { return s is string && s.matches('[A-Za-z0-9]{20}'); }

    // ---------- Owner setup ----------
    // config/pins {setup} is made by hand in the Firebase console just before first-time setup. No app can read it.
    // Setup is one batch: config/owner {uid, proof}, the owner's staff and username records, an audit entry and
    // the deletion of config/pins. config/owner can never be changed or deleted, so only one owner can ever be
    // claimed, even if two phones try at once, and the setup code is dead the moment setup succeeds.
    match /config/{id} {
      allow read, update: if false;
      allow create: if id == 'owner' && signedIn() && exactly(['uid', 'proof'])
        && incoming().uid == request.auth.uid
        && incoming().proof is string && incoming().proof.size() >= 16
        && incoming().proof == get(/databases/$(database)/documents/config/pins).data.setup
        && getAfter(/databases/$(database)/documents/config/pins) == null
        && getAfter(staffDoc(request.auth.uid)).data.role == 'owner';
      allow delete: if id == 'pins' && signedIn() && getAfter(ownerDoc()).data.uid == request.auth.uid;
    }

    // ---------- Team accounts ----------
    // An account is three things kept in step by one batched write each time: the Firebase Auth login,
    // staff/{uid} (membership and role) and usernames/{username} (the login lookup). Every add, PIN reset
    // and removal also writes an audit entry, which these rules require.
    function validStaff() {
      return exactly(['name', 'username', 'role']) && reqText(incoming().name, 40)
        && incoming().username is string && incoming().username.matches('[a-z0-9.]{2,20}')
        && incoming().role in ['staff', 'manager', 'owner'];
    }
    match /staff/{uid} {
      allow get: if signedIn() && (request.auth.uid == uid || isManager());
      allow list: if isManager();
      allow create: if validStaff()
        && getAfter(nameDoc(incoming().username)).data.uid == uid
        && existsAfter(auditDoc(uid + '-in'))
        && ((isOwner() && incoming().role in ['staff', 'manager'])
          || (isManager() && incoming().role == 'staff')
          // First-time setup: the owner's own record, in the same write that claims config/owner.
          || (request.auth.uid == uid && incoming().role == 'owner' && getAfter(ownerDoc()).data.uid == uid));
      // Only the owner changes a role, never their own, and never to or from owner.
      allow update: if isOwner() && request.auth.uid != uid && changesOnly(['role'])
        && resource.data.role in ['staff', 'manager'] && incoming().role in ['staff', 'manager'];
      // Managers remove staff; the owner removes staff and managers; nobody removes the owner or themselves.
      // The username goes too, or moves to the new account on a PIN reset, in the same write.
      allow delete: if request.auth.uid != uid && resource.data.role != 'owner'
        && (isOwner() || (isManager() && resource.data.role == 'staff'))
        && (getAfter(nameDoc(resource.data.username)) == null || getAfter(nameDoc(resource.data.username)).data.uid != uid)
        && existsAfter(auditDoc(uid + '-out'));
    }

    // Login looks up a username before signing in, so a single lookup is public. Listing them is not.
    // A record holds the account's made-up sign-in address (never a real email) and its uid.
    function validLogin(username) {
      return username.matches('[a-z0-9.]{2,20}') && exactly(['email', 'uid'])
        && incoming().email is string && incoming().email.matches('[a-z0-9.]{2,20}-[a-z0-9]{6,12}@users[.]example[.]com')
        && incoming().uid is string
        && getAfter(staffDoc(incoming().uid)).data.username == username;
    }
    match /usernames/{username} {
      allow get: if true;
      allow create: if validLogin(username)
        && (isOwner()
          || (isManager() && getAfter(staffDoc(incoming().uid)).data.role == 'staff')
          || (incoming().uid == request.auth.uid && getAfter(staffDoc(request.auth.uid)).data.role == 'owner'
              && getAfter(ownerDoc()).data.uid == request.auth.uid));
      // PIN reset: the username moves to a new account and the old account is removed in the same write.
      allow update: if validLogin(username) && getAfter(staffDoc(resource.data.uid)) == null
        && (isOwner() || (isManager() && getAfter(staffDoc(incoming().uid)).data.role == 'staff'));
      // Removal: the account goes in the same write. The staff rules decide who may remove whom.
      allow delete: if isManager() && getAfter(staffDoc(resource.data.uid)) == null;
    }

    // ---------- Audit of account changes ----------
    // {uid}-in when an account joins (setup, add, PIN reset), {uid}-out when it leaves (removal, PIN reset),
    // and an automatic ID for a role change. Each entry must match a change made in the same write.
    // Nobody can edit or delete an entry.
    function joins(t) { return get(staffDoc(t)) == null && getAfter(staffDoc(t)) != null; }
    function leaves(t) { return get(staffDoc(t)) != null && getAfter(staffDoc(t)) == null; }
    function describes(s) {
      return incoming().targetName == s.name && incoming().username == s.username && incoming().role == s.role;
    }
    match /audit/{id} {
      allow read: if isManager();
      allow create: if signedIn()
        && exactly(['action', 'target', 'targetName', 'username', 'role', 'uid', 'name', 'at'])
        && incoming().target is string && incoming().uid == request.auth.uid && incoming().at == request.time
        // Read after the write, so this also covers the owner's first-time setup.
        && getAfter(staffDoc(request.auth.uid)).data.role in ['manager', 'owner']
        && incoming().name == getAfter(staffDoc(request.auth.uid)).data.name
        && ((id == incoming().target + '-in' && joins(incoming().target) && describes(getAfter(staffDoc(incoming().target)).data)
              && incoming().action in (incoming().role == 'owner' ? ['setup'] : ['add', 'reset']))
          || (id == incoming().target + '-out' && leaves(incoming().target) && describes(get(staffDoc(incoming().target)).data)
              && incoming().action in ['remove', 'reset'])
          || (incoming().action == 'role' && autoId(id) && get(staffDoc(incoming().target)) != null
              && get(staffDoc(incoming().target)).data.role != getAfter(staffDoc(incoming().target)).data.role
              && describes(getAfter(staffDoc(incoming().target)).data)));
    }

    // ---------- Team board and broadcasts ----------
    // The board is read-only since team chat (26 October 2026): nothing new is posted, and old posts stay until taken down.
    match /feed/{id} {
      allow read: if isMember();
      // The author or a manager can take a post down. Only the owner takes down the owner's posts.
      allow delete: if isMember() && (resource.data.uid == request.auth.uid || isOwner()
        || (isManager() && notOwner(resource.data.uid)));
    }
    // Broadcasts (stored as notices). They can't be edited: send a new one and remove the old.
    // side is who it's for: everyone, the bar or the floor. Each phone shows the ones for the side it's set to, which is
    // about relevance, not privacy. The owner can leave managers out (managers false), and then no manager can read it,
    // so a manager's phone asks only for the ones with managers true. (No default value for a missing field here: with
    // one, a query that asks for everything gets through. Broadcasts from before audiences existed are for staff and
    // the owner only.)
    match /notices/{id} {
      allow read: if isMember() && (resource.data.managers == true || profile().data.role != 'manager');
      allow create: if isManager() && exactly(['text', 'side', 'managers', 'uid', 'name', 'at']) && reqText(incoming().text, 1000)
        && incoming().side in ['all', 'bar', 'floor'] && incoming().managers is bool && (incoming().managers || isOwner()) && stamped();
      allow delete: if isOwner() || (isManager() && notOwner(resource.data.uid));
    }
    // "Got it" on a broadcast: one record per person per broadcast, so the team can see who has read it. It copies the
    // broadcast's managers setting, so managers can't see who read one they were left out of. Records can't be changed.
    match /seen/{id} {
      allow read: if isMember() && (resource.data.uid == request.auth.uid || isOwner() || (isManager() && resource.data.managers == true));
      allow create: if isMember() && exactly(['notice', 'managers', 'uid', 'name', 'at']) && stamped()
        && incoming().notice is string && id == incoming().notice + '-' + request.auth.uid
        && incoming().managers == get(/databases/$(database)/documents/notices/$(incoming().notice)).data.get('managers', true);
    }

    // ---------- Shift reports and closing photos ----------
    // A report is sent once, with its photos attached in the same write, then signed off once by a manager.
    // Nobody can edit or delete a report from the app.
    match /reports/{id} {
      allow read: if isMember();
      allow create: if isMember() && exactly(['side', 'text', 'summary', 'photos', 'uid', 'name', 'at'])
        && incoming().side in ['bar', 'floor']
        && reqText(incoming().text, 8000) && reqText(incoming().summary, 200)
        && incoming().photos is int && incoming().photos >= 0 && incoming().photos <= 8
        && stamped();
      allow update: if isManager() && !('signedBy' in resource.data)
        && changesOnly(['signedBy', 'signedByUid', 'signedAt']) && signedAs('signedBy', 'signedByUid', 'signedAt');
    }
    function reportDoc(id) { return /databases/$(database)/documents/reports/$(id); }
    match /photos/{id} {
      // Your own drafts, photos on a sent report, and (for the 30-day clear-out) managers see everything.
      allow read: if isManager() || (isMember() && (resource.data.uid == request.auth.uid || resource.data.report != null));
      allow create: if isMember() && exactly(['data', 'note', 'report', 'uid', 'name', 'at'])
        && incoming().data is string && incoming().data.size() <= 700000
        && incoming().data.matches('data:image/jpeg;base64,[A-Za-z0-9+/=]+')
        && incoming().note == '' && incoming().report == null && stamped();
      // Until it's sent, the person who took it can change the note. It is attached only inside the write that
      // creates their report (the report's time is this request's time), so it can't move onto an older report.
      allow update: if isMember() && resource.data.uid == request.auth.uid && resource.data.report == null
        && ((changesOnly(['note']) && optText(incoming().note, 300))
          || (changesOnly(['report']) && autoId(incoming().report)
              && getAfter(reportDoc(incoming().report)).data.uid == request.auth.uid
              && getAfter(reportDoc(incoming().report)).data.at == request.time));
      // Drafts can be binned by whoever took them. Anything over 30 days old is cleared by a manager's app.
      allow delete: if isMember() && ((resource.data.uid == request.auth.uid && resource.data.report == null)
        || (isManager() && resource.data.at < request.time - duration.value(30, 'd')));
    }

    // ---------- Tonight's checklists: tasks and procedures ----------
    // {day} is the business day (YYYY-MM-DD, rolling over at 6am UK time). Writes are only accepted within a day or two
    // of it, which allows for a phone that was offline and syncs late.
    function nearNow(day) {
      return isDay(day) && dayOf(day) > request.time - duration.value(2, 'd') && dayOf(day) < request.time + duration.value(1, 'd');
    }
    // Each list is a procedure: the bar's opening and closing, the floor's tables, milkshake station and close.
    function isList(l) { return l in ['open', 'close', 'fset', 'fshake', 'fclose']; }
    function sideOf(l) { return l in ['open', 'close'] ? 'bar' : 'floor'; }
    function taskDoc(day, id) { return /databases/$(database)/documents/shift/$(day)/tasks/$(id); }
    function runDoc(day, list) { return /databases/$(database)/documents/shift/$(day)/runs/$(list); }
    function eventDoc(id) { return /databases/$(database)/documents/events/$(id); }
    // A field as it was before this write, or d for a new record.
    function was(k, d) { return resource == null ? d : resource.data[k]; }
    // Someone on the team, named as on their staff record, or nobody (both null).
    function teammate(name, uid) { return (name == null && uid == null) || (uid is string && get(staffDoc(uid)).data.name == name); }
    // A procedure is locked once it's finished, until a manager reopens it.
    function unlocked(day, list) { let r = get(runDoc(day, list)); return r == null || r.data.status != 'finished'; }

    // Ticks from before tasks (24 and 25 October 2026), kept for the record. Nothing new is written.
    match /shift/{day}/ticks/{id} {
      allow read: if isMember();
    }

    // One record per item per night: todo (not started), doing, done, blocked or cancelled (not needed tonight). Each
    // change raises v by one and writes events/t-{day}-{list}-{item}-{v} in the same batch, so a change can't happen
    // without its event, an event can't be made up, and a phone working from an old copy (two people at once, or a late
    // sync) is refused instead of overwriting the newer change. Nobody deletes a task.
    match /shift/{day}/tasks/{id} {
      allow read: if isMember();
      allow create, update: if isMember() && nearNow(day) && stamped() && validTask(day, id)
        && existsAfter(eventDoc('t-' + day + '-' + id + '-' + string(incoming().v)));
    }
    function validTask(day, id) {
      let t = incoming();
      let from = was('status', 'todo');
      return exactly(['list', 'item', 'status', 'who', 'whoUid', 'note', 'v', 'uid', 'name', 'at'])
        && isList(t.list) && t.item is int && t.item >= 0 && t.item < 60 && id == t.list + '-' + string(t.item)
        && t.v == was('v', 0) + 1 && optText(t.note, 200) && unlocked(day, t.list)
        && ((t.status == from && givenTo(t, from))
          || (t.status != from && t.who == was('who', null) && t.whoUid == was('whoUid', null) && taskMove(from, t.status, t.note)));
    }
    // Managers give a task to someone, or take it off them, without changing where it's up to.
    function givenTo(t, from) {
      return isManager() && from in ['todo', 'doing', 'blocked'] && t.whoUid != was('whoUid', null) && teammate(t.who, t.whoUid);
    }
    // Where a task can go from where it is. Blocked and Not needed need a reason, and so does reopening a done task,
    // unless whoever did it takes it back within 5 minutes (a mis-tap). Only managers mark a task not needed.
    function taskMove(from, to, note) {
      let mine = was('uid', '') == request.auth.uid;
      return (to == 'doing' && from in ['todo', 'blocked'])
        || (to == 'done' && from in ['todo', 'doing', 'blocked'])
        || (to == 'blocked' && from in ['todo', 'doing'] && reqText(note, 200))
        || (to == 'cancelled' && from in ['todo', 'doing', 'blocked'] && isManager() && reqText(note, 200))
        || (to == 'todo' && from == 'blocked')
        || (to == 'todo' && from == 'doing' && (mine || isManager()))
        || (to == 'todo' && from == 'done' && (mine || isManager())
          && (reqText(note, 200) || (mine && request.time < resource.data.at + duration.value(5, 'm'))))
        || (to == 'todo' && from == 'cancelled' && isManager());
    }

    // One record per list per night: started, then finished. Finishing with tasks left needs a manager and a reason
    // (an override), and so does reopening. done and required are counted by the phone that finishes it: the rules
    // can't count the tasks, so the tasks and their events stay the record of what was done.
    match /shift/{day}/runs/{list} {
      allow read: if isMember();
      allow create, update: if isMember() && nearNow(day) && stamped() && validRun(list)
        && existsAfter(eventDoc('r-' + day + '-' + list + '-' + string(incoming().v)));
    }
    function validRun(list) {
      let r = incoming();
      let from = was('status', 'todo');
      return exactly(['list', 'status', 'done', 'required', 'note', 'v', 'uid', 'name', 'at'])
        && r.list == list && isList(list) && r.v == was('v', 0) + 1 && optText(r.note, 300)
        && r.done is int && r.required is int && r.done >= 0 && r.done <= r.required && r.required <= 60
        && ((r.status == 'started' && from == 'todo')
          || (r.status == 'finished' && from in ['todo', 'started'] && (r.done == r.required || (isManager() && reqText(r.note, 300))))
          || (r.status == 'started' && from == 'finished' && isManager() && reqText(r.note, 300)));
    }

    // ---------- Restocking from the cellar ----------
    // Someone asks for an item, an amount and where it's going, with any detail (which never changes). Someone takes it
    // on (or a manager gives it to someone), brings it up saying how much they moved, and the bar or floor confirms how
    // much arrived. Asked, moved and received are kept apart, so a request is never taken as proof that the stock
    // arrived, and no stock level changes here. note is the reason for the latest change. Each change raises v and
    // writes events/s-{id}-{v} in the same batch, like tasks.
    function restockDoc(id) { return /databases/$(database)/documents/restock/$(id); }
    function amount(n) { return n is number && n > 0 && n <= 1000; }
    match /restock/{id} {
      allow read: if isMember();
      allow create: if isMember() && stamped() && autoId(id) && newRestock() && existsAfter(eventDoc('s-' + id + '-1'));
      allow update: if isMember() && stamped() && restockMove() && existsAfter(eventDoc('s-' + id + '-' + string(incoming().v)));
    }
    function newRestock() {
      let r = incoming();
      return exactly(['item', 'qty', 'unit', 'dest', 'detail', 'day', 'status', 'who', 'whoUid', 'moved', 'got', 'note', 'v', 'askedBy', 'askedUid', 'askedAt', 'uid', 'name', 'at'])
        && reqText(r.item, 80) && amount(r.qty) && r.unit in ['each', 'bottle', 'case', 'box', 'pack', 'bag', 'roll', 'keg', 'tub']
        && r.dest in ['up', 'down', 'shakes', 'floor'] && nearNow(r.day) && optText(r.detail, 200) && r.note == ''
        && r.status == 'asked' && r.who == null && r.whoUid == null && r.moved == null && r.got == null && r.v == 1
        && r.askedBy == profile().data.name && r.askedUid == request.auth.uid && r.askedAt == request.time;
    }
    function restockMove() {
      let r = incoming();
      let old = resource.data;
      let to = r.status;
      let from = old.status;
      let mine = old.whoUid == request.auth.uid;
      let me = r.whoUid == request.auth.uid && r.who == profile().data.name;
      let same = r.who == old.who && r.whoUid == old.whoUid && r.moved == old.moved && r.got == old.got;
      return r.diff(old).affectedKeys().hasOnly(['status', 'who', 'whoUid', 'moved', 'got', 'note', 'v', 'uid', 'name', 'at'])
        && r.v == old.v + 1 && optText(r.note, 200) && (
          // Given to someone, or to someone else, by a manager.
          (to == 'given' && from in ['asked', 'given', 'doing', 'blocked'] && isManager() && r.whoUid != null && r.whoUid != old.whoUid
            && teammate(r.who, r.whoUid) && r.moved == null && r.got == null)
          // Taken on by anyone while it's waiting; once it's been given to someone, by them or a manager.
          || (to == 'doing' && from in ['asked', 'given', 'blocked'] && me && (from != 'given' || mine || isManager()) && r.moved == null && r.got == null)
          // Brought up by whoever has it (or a manager), saying how much was moved.
          || (to == 'delivered' && from == 'doing' && (mine || isManager()) && r.who == old.who && r.whoUid == old.whoUid && amount(r.moved) && r.got == null)
          // Confirmed by whoever is there, saying how much arrived.
          || (to == 'received' && from == 'delivered' && r.who == old.who && r.whoUid == old.whoUid && r.moved == old.moved && amount(r.got))
          || (to == 'unavailable' && from in ['asked', 'given', 'doing', 'blocked'] && same)
          || (to == 'blocked' && from in ['asked', 'given', 'doing'] && same && reqText(r.note, 200))
          || (to == 'cancelled' && from in ['asked', 'given', 'doing', 'blocked'] && same
            && (old.askedUid == request.auth.uid || (isManager() && reqText(r.note, 200))))
          // Back to waiting: let go by whoever has it (or a manager), not there after all once brought up (with a
          // reason), or reopened by a manager (with a reason).
          || (to == 'asked' && r.who == null && r.whoUid == null && r.moved == null && r.got == null
            && ((from in ['given', 'doing'] && (mine || isManager()))
              || (from == 'delivered' && reqText(r.note, 200))
              || (from in ['received', 'unavailable', 'cancelled'] && isManager() && reqText(r.note, 200)))));
    }

    // ---------- Audit trail ----------
    // One event per change to a task, a procedure or a restock, written in the same batch as the change and matching it
    // (the status before and after, the version, who, the reason and the time; for a restock, the item, amount and place). Nobody edits or deletes an event, the
    // owner included: a correction is a new change with its own event. Ids carry the record's version, so a retried
    // or repeated write can't add a second event for the same change.
    match /events/{id} {
      allow read: if isMember();
      allow create: if isMember() && stamped() && nearNow(incoming().day) && reqText(incoming().label, 200) && optText(incoming().note, 300)
        && (incoming().kind == 'task' ? taskEvent(id) : incoming().kind == 'run' ? runEvent(id) : incoming().kind == 'restock' && restockEvent(id));
    }
    function taskEvent(id) {
      let e = incoming();
      let path = taskDoc(e.day, e.list + '-' + string(e.item));
      let now = getAfter(path).data;
      let before = get(path);
      return exactly(['kind', 'day', 'list', 'item', 'label', 'area', 'from', 'to', 'who', 'note', 'v', 'uid', 'name', 'at'])
        && e.item is int && id == 't-' + e.day + '-' + e.list + '-' + string(e.item) + '-' + string(e.v)
        && now.at == request.time && now.v == e.v && now.status == e.to && now.who == e.who && now.note == e.note
        && e.from == (before == null ? 'todo' : before.data.status) && e.area == sideOf(e.list);
    }
    function runEvent(id) {
      let e = incoming();
      let path = runDoc(e.day, e.list);
      let now = getAfter(path).data;
      let before = get(path);
      return exactly(['kind', 'day', 'list', 'label', 'area', 'from', 'to', 'done', 'required', 'note', 'v', 'uid', 'name', 'at'])
        && id == 'r-' + e.day + '-' + e.list + '-' + string(e.v)
        && now.at == request.time && now.v == e.v && now.status == e.to && now.note == e.note && now.done == e.done && now.required == e.required
        && e.from == (before == null ? 'todo' : before.data.status) && e.area == sideOf(e.list);
    }
    function restockEvent(id) {
      let e = incoming();
      let path = restockDoc(e.ref);
      let now = getAfter(path).data;
      let before = get(path);
      return exactly(['kind', 'day', 'ref', 'label', 'area', 'from', 'to', 'who', 'qty', 'unit', 'dest', 'note', 'v', 'uid', 'name', 'at'])
        && e.ref is string && id == 's-' + e.ref + '-' + string(e.v)
        && now.at == request.time && now.v == e.v && now.status == e.to && now.who == e.who && now.note == e.note
        && now.item == e.label && now.unit == e.unit && now.dest == e.dest
        && e.from == (before == null ? 'new' : before.data.status)
        && e.qty == (e.to == 'delivered' ? now.moved : e.to == 'received' ? now.got : e.from == 'new' ? now.qty : null)
        && e.area == (e.dest in ['up', 'down'] ? 'bar' : 'floor');
    }

    // ---------- Old records: waste and due diligence ----------
    // Kept read-only as the venue's record since 25 Oct 2026. Temperatures and line cleans are logged in Alert65 and
    // waste in Nory now, so nothing new is written here and nothing is edited or deleted. Managers download them.
    match /waste/{id} {
      allow read: if isManager() || (isMember() && resource.data.uid == request.auth.uid);
    }
    match /kit/{id} {
      allow read: if isMember();
    }
    match /safety/{id} {
      allow read: if isMember();
    }

    // ---------- 86 board ----------
    // out/{item} is what's off tonight and outlog is its history. Each change writes both in one batch and the
    // rules check they match: an 86 creates out/{item} pointing at its new log entry; back in stock deletes it
    // and adds the entry {that log id}-back. So the board and its history can't drift apart, the same item
    // can't be 86'd twice, and nothing comes back that wasn't out. Log entries can never be edited or deleted.
    function outDoc(item) { return /databases/$(database)/documents/out/$(item); }
    function logDoc(id) { return /databases/$(database)/documents/outlog/$(id); }
    match /out/{item} {
      allow read: if isMember();
      allow create: if isMember() && exactly(['itemName', 'log', 'uid', 'name', 'at'])
        && reqText(incoming().itemName, 80) && slug(incoming().itemName) == item && autoId(incoming().log)
        && getAfter(logDoc(incoming().log)).data.action == 'out' && getAfter(logDoc(incoming().log)).data.item == item
        && stamped();
      allow delete: if isMember() && existsAfter(logDoc(resource.data.log + '-back'));
    }
    match /outlog/{id} {
      allow read: if isManager();
      allow create: if isMember() && exactly(['item', 'itemName', 'action', 'uid', 'name', 'at']) && stamped()
        && reqText(incoming().itemName, 80) && incoming().item == slug(incoming().itemName)
        && ((incoming().action == 'out' && getAfter(outDoc(incoming().item)).data.log == id)
          || (incoming().action == 'back' && get(outDoc(incoming().item)).data.log + '-back' == id
              && getAfter(outDoc(incoming().item)) == null));
    }

    // ---------- Stock requests (before restocking from the cellar, 25 October 2026) ----------
    // Kept for the record. Nothing new is asked for here; managers can still mark old ones ordered or turn them down.
    match /requests/{id} {
      allow read: if isMember();
      // A manager marks it ordered, once. What was asked for never changes.
      allow update: if isManager() && resource.data.done == false
        && changesOnly(['done', 'doneBy', 'doneByUid', 'doneAt']) && incoming().done == true
        && signedAs('doneBy', 'doneByUid', 'doneAt');
      // Until it's ordered, the person who asked can cancel it and a manager can turn it down.
      allow delete: if isMember() && resource.data.done == false && (resource.data.uid == request.auth.uid || isManager());
    }

    // ---------- Old records: stock counts and par levels ----------
    // Read-only since 25 Oct 2026: stock is counted and ordered in Nory. Each count keeps every item's par as it was
    // then, so its order list can still be worked out from the download.
    match /pars/{id} {
      allow read: if isManager();
    }
    match /counts/{id} {
      allow read: if isManager();
    }

    // ---------- Running late, time off and shift swaps ----------
    // A real date from yesterday to about a year ahead.
    function soonDay(s) {
      return isDay(s) && dayOf(s) > request.time - duration.value(2, 'd') && dayOf(s) < request.time + duration.value(400, 'd');
    }
    function validAsk() {
      let a = incoming();
      let base = ['kind', 'note', 'status', 'uid', 'name', 'at'];
      return a.status == 'pending' && optText(a.note, 300) && stamped()
        && ((a.kind == 'late' && exactly(base.concat(['eta'])) && isTime(a.eta))
          || (a.kind == 'off' && exactly(base.concat(['date1', 'date2'])) && soonDay(a.date1) && soonDay(a.date2) && a.date2 >= a.date1)
          || (a.kind == 'swap' && exactly(base.concat(['date1', 'with'])) && soonDay(a.date1) && reqText(a.with, 40)));
    }
    match /asks/{id} {
      allow read: if isManager() || (isMember() && resource.data.uid == request.auth.uid);
      allow create: if isMember() && validAsk();
      // Seen means a manager has read it, and it can still be approved or declined afterwards.
      // Approved and declined are final. Managers don't decide their own; the owner can decide any.
      allow update: if isManager() && (resource.data.uid != request.auth.uid || isOwner())
        && changesOnly(['status', 'by', 'byUid', 'byAt']) && signedAs('by', 'byUid', 'byAt')
        && ((resource.data.status == 'pending' && incoming().status in ['seen', 'approved', 'declined'])
          || (resource.data.status == 'seen' && incoming().status in ['approved', 'declined']));
      // The person who asked can withdraw it until it's decided. The owner can delete any.
      allow delete: if isOwner() || (isMember() && resource.data.uid == request.auth.uid && resource.data.status in ['pending', 'seen']);
    }

    // ---------- Incidents ----------
    // Anyone on the team logs one; only managers read them. A manager closes it once, saying what was done.
    // The original account can't be edited. Only the owner can delete one.
    match /incidents/{id} {
      allow read: if isManager();
      allow create: if isMember() && exactly(['kind', 'text', 'status', 'uid', 'name', 'at'])
        && reqText(incoming().kind, 40) && reqText(incoming().text, 1000) && incoming().status == 'open' && stamped();
      allow update: if isManager() && resource.data.status == 'open'
        && changesOnly(['status', 'outcome', 'closedBy', 'closedByUid', 'closedAt'])
        && incoming().status == 'closed' && reqText(incoming().outcome, 500)
        && signedAs('closedBy', 'closedByUid', 'closedAt');
      allow delete: if isOwner();
    }

    // ---------- Managers' reviews of shift reports ----------
    // A thumbs up can stand alone; a flag or a comment must say something. Reviews can't be edited.
    match /reviews/{id} {
      allow read: if isManager();
      allow create: if isManager() && exactly(['report', 'verdict', 'text', 'uid', 'name', 'at']) && stamped()
        && autoId(incoming().report) && exists(reportDoc(incoming().report))
        && ((incoming().verdict == 'up' && optText(incoming().text, 500))
          || (incoming().verdict in ['flag', 'note'] && reqText(incoming().text, 500)));
      allow delete: if isOwner() || (isManager() && resource.data.uid == request.auth.uid);
    }

    // ---------- Anonymous feedback between managers ----------
    // No author is stored, so it can't be traced, withdrawn or limited per person in the app.
    // The owner reads all of it; a manager reads only what's about them. "Needs improvement" must say why.
    match /ratings/{id} {
      allow read: if isOwner() || (isManager() && resource.data.about == request.auth.uid);
      allow create: if isManager() && exactly(['about', 'aboutName', 'score', 'text', 'at'])
        && incoming().about is string && incoming().about != request.auth.uid
        && get(staffDoc(incoming().about)).data.role in ['manager', 'owner']
        && incoming().aboutName == get(staffDoc(incoming().about)).data.name
        && ((incoming().score in ['Excellent', 'Efficient', 'Good'] && optText(incoming().text, 500))
          || (incoming().score == 'Needs improvement' && reqText(incoming().text, 500)))
        && incoming().at == request.time;
      allow delete: if isOwner();
    }

    // ---------- Training ----------
    // A manager signs someone off on a drink once they've made it to spec in front of them. One record per person per
    // drink, named after both (drinks are named by their slug, like the 86 board, since the list order changes).
    // Nobody signs themselves off. Staff see their own; managers see everyone's. Taking a sign-off back deletes it.
    match /training/{id} {
      allow read: if isManager() || (isMember() && resource.data.trainee == request.auth.uid);
      allow create: if isManager() && exactly(['trainee', 'traineeName', 'drinkName', 'uid', 'name', 'at'])
        && incoming().trainee is string && incoming().trainee != request.auth.uid && reqText(incoming().drinkName, 80)
        && id == incoming().trainee + '-' + slug(incoming().drinkName)
        && get(staffDoc(incoming().trainee)).data.name == incoming().traineeName && stamped();
      allow delete: if isManager();
    }

    // ---------- Issues and handover ----------
    // Faults, jobs and anything the next shift needs to know. Open until someone fixes it, then kept as history.
    // A fixed issue can't be edited or reopened; "still broken" raises a new one that links back to it.
    match /issues/{id} {
      allow read: if isMember();
      allow create: if isMember() && exactly(['title', 'detail', 'area', 'urgent', 'again', 'status', 'uid', 'name', 'at'])
        && reqText(incoming().title, 120) && optText(incoming().detail, 1000)
        && incoming().area in ['bar', 'floor', 'kitchen', 'toilets', 'terrace', 'cellar', 'other']
        && incoming().urgent is bool && incoming().status == 'open'
        && (incoming().again == null || (autoId(incoming().again) && exists(/databases/$(database)/documents/issues/$(incoming().again))))
        && stamped();
      // Take it on yourself (managers can give it to anyone on the team), or mark it fixed saying what was done.
      allow update: if isMember() && resource.data.status in ['open', 'taken']
        && ((incoming().status == 'taken' && changesOnly(['status', 'who', 'whoUid', 'setBy', 'setByUid', 'setAt'])
              && signedAs('setBy', 'setByUid', 'setAt') && incoming().whoUid is string
              && ((incoming().whoUid == request.auth.uid && incoming().who == profile().data.name)
                || (isManager() && get(staffDoc(incoming().whoUid)).data.name == incoming().who)))
          || (incoming().status == 'fixed' && changesOnly(['status', 'fix', 'fixedBy', 'fixedByUid', 'fixedAt'])
              && reqText(incoming().fix, 500) && signedAs('fixedBy', 'fixedByUid', 'fixedAt')));
      // Whoever raised it can withdraw it before anyone takes it on. The owner can delete any.
      allow delete: if isOwner() || (isMember() && resource.data.uid == request.auth.uid && resource.data.status == 'open');
    }

    // ---------- Concerns ----------
    // Anyone on the team reports something that isn't right (needs cleaning, untidy, broken, a hazard) with up to three
    // photos, to everyone, the bar, the floor or managers only. An anonymous concern saves no uid or name, so no app
    // account can see who sent it, the sender included. Bar and floor are about relevance, like broadcasts; managers
    // only is enforced, for the concern and its photos.
    function concernDoc(id) { return /databases/$(database)/documents/concerns/$(id); }
    function validConcern() {
      let c = incoming();
      return c.reason in ['cleaning', 'unkempt', 'broken', 'hazard', 'other']
        && c.area in ['bar', 'floor', 'kitchen', 'toilets', 'terrace', 'cellar', 'other']
        && optText(c.note, 300) && (c.reason != 'other' || reqText(c.note, 300))
        && c.to in ['all', 'bar', 'floor', 'managers'] && c.photos is int && c.photos >= 0 && c.photos <= 3
        && c.status == 'open' && c.at == request.time
        && ((c.anon == true && exactly(['reason', 'area', 'note', 'to', 'photos', 'anon', 'status', 'at']))
          || (c.anon == false && exactly(['reason', 'area', 'note', 'to', 'photos', 'anon', 'status', 'uid', 'name', 'at']) && stamped()));
    }
    match /concerns/{id} {
      allow read: if isManager() || (isMember() && (resource.data.to in ['all', 'bar', 'floor'] || resource.data.uid == request.auth.uid));
      allow create: if isMember() && validConcern();
      // Dealt with, once, with an optional note. Anyone can for a concern sent to the team; managers only ones need a manager.
      allow update: if isMember() && resource.data.status == 'open' && (isManager() || resource.data.to in ['all', 'bar', 'floor'])
        && incoming().status == 'closed' && changesOnly(['status', 'fix', 'fixedBy', 'fixedByUid', 'fixedAt'])
        && optText(incoming().fix, 300) && signedAs('fixedBy', 'fixedByUid', 'fixedAt');
      // Managers can take any concern down. Whoever sent a named one can withdraw it while it's open.
      allow delete: if isManager() || (isMember() && resource.data.get('uid', '') == request.auth.uid && resource.data.status == 'open');
    }
    // Photos are added only in the write that creates their concern, numbered from 0, and carry its audience so
    // managers only photos stay with managers. They go when their concern goes, or after 30 days (a manager's app clears them).
    match /concernPhotos/{id} {
      allow read: if isManager() || (isMember() && resource.data.to in ['all', 'bar', 'floor']);
      allow create: if isMember() && exactly(['concern', 'n', 'to', 'data', 'at'])
        && incoming().concern is string && incoming().n is int && incoming().n >= 0
        && id == incoming().concern + '-' + string(incoming().n)
        && !exists(concernDoc(incoming().concern))
        && getAfter(concernDoc(incoming().concern)).data.photos > incoming().n
        && getAfter(concernDoc(incoming().concern)).data.to == incoming().to
        && incoming().data is string && incoming().data.size() <= 700000
        && incoming().data.matches('data:image/jpeg;base64,[A-Za-z0-9+/=]+') && incoming().at == request.time;
      // (Removing an old concern also names photos that were already cleared; deleting nothing is allowed.)
      allow delete: if isMember() && (resource == null || !existsAfter(concernDoc(resource.data.concern))
        || (isManager() && resource.data.at < request.time - duration.value(30, 'd')));
    }

    // ---------- Team chat ----------
    // Who's on the team, for picking someone to message: name and role only. Staff can't list staff records (they hold
    // usernames), so each phone keeps its own entry up to date and managers' phones keep everyone's. An entry always
    // matches its staff record, and goes when the person does.
    match /directory/{uid} {
      allow read: if isMember();
      allow create, update: if isMember() && (uid == request.auth.uid || isManager()) && exactly(['name', 'role'])
        && get(staffDoc(uid)).data.name == incoming().name && get(staffDoc(uid)).data.role == incoming().role;
      allow delete: if isManager() && get(staffDoc(uid)) == null;
    }
    // Three kinds of chat: the team chat (id team, everyone on the team is in it), private chats between two people (one
    // per pair, named after both uids in order) and groups (anyone starts one, with a name and who's in it). Only the
    // people in a private chat or a group can read it: managers and the owner can't read other people's chats in the app.
    // A chat keeps a preview of its latest message (with who it pinged), written in the same batch as the message, and
    // read {uid: time}, when each person last read it, which is how a phone shows what's unread.
    function chatDoc(id) { return /databases/$(database)/documents/chats/$(id); }
    function msgDoc(chat, id) { return /databases/$(database)/documents/chats/$(chat)/msgs/$(id); }
    function chatPhotoDoc(id) { return /databases/$(database)/documents/chatPhotos/$(id); }
    function inChat(c) { return c.kind == 'team' || request.auth.uid in c.members; }
    // Marks only the caller's own read time, at the server's clock.
    function readNow(c) {
      return c.read.diff(resource.data.read).affectedKeys().hasOnly([request.auth.uid]) && c.read[request.auth.uid] == request.time;
    }
    match /chats/{id} {
      // Asking for the team chat before anyone has made it is allowed (there's nothing to see). A list must ask only for
      // chats the caller is in.
      allow get: if isMember() && (resource == null || inChat(resource.data));
      allow list: if isMember() && request.auth.uid in resource.data.members;
      allow create: if isMember() && newChat(id);
      allow update: if isMember() && inChat(resource.data) && (
        (changesOnly(['read']) && readNow(incoming()))
        || newMessage(id)
        // Unsending your own latest message clears the preview.
        || (changesOnly(['lastText']) && incoming().lastText == '' && resource.data.lastUid == request.auth.uid
          && !existsAfter(msgDoc(id, resource.data.lastId)))
        || groupChange());
    }
    function newChat(id) {
      let c = incoming();
      return exactly(['kind', 'title', 'members', 'by', 'at', 'lastId', 'lastText', 'lastName', 'lastUid', 'lastAt', 'lastPings', 'read'])
        && c.by == request.auth.uid && c.at == request.time && c.lastAt == request.time
        && c.lastId == '' && c.lastText == '' && c.lastName == '' && c.lastUid == '' && c.lastPings == [] && c.read == {}
        && ((c.kind == 'team' && id == 'team' && c.title == '' && c.members == [])
          || (c.kind == 'dm' && c.title == '' && c.members is list && c.members.size() == 2 && request.auth.uid in c.members
            && c.members[0] is string && c.members[1] is string && c.members[0] < c.members[1] && id == c.members[0] + '-' + c.members[1]
            && get(staffDoc(c.members[0])) != null && get(staffDoc(c.members[1])) != null)
          || (c.kind == 'group' && autoId(id) && reqText(c.title, 40) && c.members is list
            && c.members.size() >= 2 && c.members.size() <= 40 && request.auth.uid in c.members));
    }
    // The preview of a message the caller sends in the same batch. Sending also marks the chat read for them.
    function newMessage(id) {
      let c = incoming();
      let m = getAfter(msgDoc(id, c.lastId)).data;
      return changesOnly(['lastId', 'lastText', 'lastName', 'lastUid', 'lastAt', 'lastPings', 'read'])
        && c.lastUid == request.auth.uid && c.lastName == profile().data.name && c.lastAt == request.time && optText(c.lastText, 200)
        && readNow(c) && m.uid == request.auth.uid && m.at == request.time && c.lastPings == m.mentions;
    }
    // Whoever started a group, or a manager in it, renames it and changes who's in it (staying in it themselves).
    // Anyone can leave a group.
    function groupChange() {
      let c = incoming();
      return resource.data.kind == 'group' && changesOnly(['title', 'members']) && reqText(c.title, 40)
        && c.members is list && c.members.size() <= 40
        && (((resource.data.by == request.auth.uid || isManager()) && request.auth.uid in c.members && c.members.size() >= 2)
          || (changesOnly(['members']) && c.members == resource.data.members.removeAll([request.auth.uid])));
    }
    // Messages can't be edited. Whoever sent one can unsend it, and its photo goes with it. Up to five people are pinged
    // (mentioned) per message, and they must be in the chat; in a private chat the other person is pinged every time.
    match /chats/{chat}/msgs/{id} {
      allow read: if isMember() && inChat(get(chatDoc(chat)).data);
      allow create: if isMember() && autoId(id) && stamped() && exactly(['text', 'photo', 'mentions', 'uid', 'name', 'at'])
        && optText(incoming().text, 1000) && incoming().photo is bool && (incoming().photo || incoming().text.trim().size() > 0)
        && incoming().mentions is list && incoming().mentions.size() <= 5 && !(request.auth.uid in incoming().mentions)
        && inChat(get(chatDoc(chat)).data)
        && (get(chatDoc(chat)).data.kind == 'team' || incoming().mentions.hasOnly(get(chatDoc(chat)).data.members))
        && getAfter(chatDoc(chat)).data.lastId == id
        && (!incoming().photo || existsAfter(chatPhotoDoc(chat + '-' + id)));
      allow delete: if isMember() && resource.data.uid == request.auth.uid
        && (!resource.data.photo || !existsAfter(chatPhotoDoc(chat + '-' + id)));
    }
    // A message's photo, kept apart so the chat loads quickly, added only in the write that sends its message. Readable by
    // the people in that chat. day is the business day it was sent: the sender's phone clears their own after 30 days.
    match /chatPhotos/{id} {
      allow read: if isMember() && (resource.data.uid == request.auth.uid || inChat(get(chatDoc(resource.data.chat)).data));
      allow create: if isMember() && stamped() && exactly(['chat', 'msg', 'day', 'data', 'uid', 'name', 'at'])
        && incoming().chat is string && incoming().msg is string && id == incoming().chat + '-' + incoming().msg
        && getAfter(msgDoc(incoming().chat, incoming().msg)).data.uid == request.auth.uid
        && getAfter(msgDoc(incoming().chat, incoming().msg)).data.at == request.time
        && getAfter(msgDoc(incoming().chat, incoming().msg)).data.photo == true
        && nearNow(incoming().day) && incoming().data is string && incoming().data.size() <= 700000
        && incoming().data.matches('data:image/jpeg;base64,[A-Za-z0-9+/=]+');
      // (Unsending a message names its photo even if it was already cleared; deleting nothing is allowed.)
      allow delete: if isMember() && (resource == null || resource.data.uid == request.auth.uid);
    }
    // Pings: one record per person per chat, {to}-{chat}, pointing at the latest message that pinged them, written in the
    // same batch as that message. Only the person pinged reads it, and their phone deletes it once they've read the chat.
    match /pings/{id} {
      allow read, delete: if isMember() && resource.data.to == request.auth.uid;
      allow create, update: if isMember() && stamped() && exactly(['to', 'chat', 'msg', 'text', 'uid', 'name', 'at'])
        && incoming().to is string && incoming().chat is string && incoming().msg is string
        && id == incoming().to + '-' + incoming().chat && optText(incoming().text, 200)
        && getAfter(msgDoc(incoming().chat, incoming().msg)).data.uid == request.auth.uid
        && getAfter(msgDoc(incoming().chat, incoming().msg)).data.at == request.time
        && incoming().to in getAfter(msgDoc(incoming().chat, incoming().msg)).data.mentions
        && get(staffDoc(incoming().to)) != null;
    }
  }
