rules_version = '2';
// Black Tap Bar Book live team feed.
// Access comes from a staff/{uid} document. Roles: owner (Derek, one account), manager, staff.
// The owner manages everyone; managers can add, reset and remove staff only.
// config/pins.setup is a one-off code for creating the owner account. Nobody can read config; only these rules can.
service cloud.firestore {
  match /databases/{database}/documents {
    function signedIn() { return request.auth != null; }
    function me() { return get(/databases/$(database)/documents/staff/$(request.auth.uid)).data; }
    function isMember() { return signedIn() && exists(/databases/$(database)/documents/staff/$(request.auth.uid)); }
    function isManager() { return isMember() && me().role in ['manager', 'owner']; }
    function isOwner() { return isMember() && me().role == 'owner'; }
    function roleOf(uid) { return get(/databases/$(database)/documents/staff/$(uid)).data.role; }
    function roleAfter(uid) { return getAfter(/databases/$(database)/documents/staff/$(uid)).data.role; }
    function text(s, max) { return s is string && s.size() > 0 && s.size() <= max; }
    function validStaff() {
      let d = request.resource.data;
      return d.keys().hasOnly(['name', 'username', 'role', 'setup']) && text(d.name, 40)
        && d.username is string && d.username.matches('^[a-z0-9.]{2,20}$') && d.role in ['staff', 'manager', 'owner'];
    }

    match /config/{id} { allow read, write: if false; }

    // Login looks up a username before signing in, so single lookups are public. Listing is not.
    // Managers can only point usernames at staff accounts (adding and PIN resets).
    match /usernames/{username} {
      allow get: if true;
      allow create, update: if username.matches('^[a-z0-9.]{2,20}$')
        && request.resource.data.keys().hasOnly(['email', 'uid']) && request.resource.data.email is string && request.resource.data.uid is string
        && (isOwner() || (isManager() && roleAfter(request.resource.data.uid) == 'staff'
              && (resource == null || roleOf(resource.data.uid) == 'staff')));
      allow delete: if isOwner() || (isManager() && roleOf(resource.data.uid) == 'staff');
    }

    match /staff/{uid} {
      allow get: if signedIn() && (request.auth.uid == uid || isManager());
      allow list: if isManager();
      allow create: if validStaff() && (
        (isOwner() && request.resource.data.role in ['staff', 'manager'])
        || (isManager() && request.resource.data.role == 'staff')
        || (request.auth.uid == uid && request.resource.data.role == 'owner'
            && request.resource.data.setup == get(/databases/$(database)/documents/config/pins).data.setup));
      // Only the owner changes roles, and never their own.
      allow update: if isOwner() && validStaff() && request.auth.uid != uid
        && resource.data.role != 'owner' && request.resource.data.role in ['staff', 'manager'];
      allow delete: if request.auth.uid != uid && resource.data.role != 'owner'
        && (isOwner() || (isManager() && resource.data.role == 'staff'));
    }

    match /feed/{id} {
      allow read: if isMember();
      allow create: if isMember()
        && request.resource.data.keys().hasOnly(['kind', 'text', 'uid', 'name', 'at'])
        && request.resource.data.kind == 'comment'
        && text(request.resource.data.text, 2000)
        && request.resource.data.uid == request.auth.uid
        && request.resource.data.name == me().name
        && request.resource.data.at == request.time;
      allow delete: if isManager() || (isMember() && resource.data.uid == request.auth.uid);
    }

    // Pinned notices: managers write and remove them, everyone reads.
    match /notices/{id} {
      allow read: if isMember();
      allow create: if isManager()
        && request.resource.data.keys().hasOnly(['text', 'uid', 'name', 'at'])
        && text(request.resource.data.text, 1000)
        && request.resource.data.uid == request.auth.uid
        && request.resource.data.name == me().name
        && request.resource.data.at == request.time;
      allow delete: if isManager();
    }

    match /reports/{id} {
      allow read: if isMember();
      allow create: if isMember()
        && request.resource.data.keys().hasOnly(['text', 'summary', 'photos', 'uid', 'name', 'at'])
        && text(request.resource.data.text, 4000)
        && text(request.resource.data.summary, 200)
        && request.resource.data.photos is int
        && request.resource.data.uid == request.auth.uid
        && request.resource.data.name == me().name
        && request.resource.data.at == request.time;
      // Managers sign off a shift report once.
      allow update: if isManager() && !('signedBy' in resource.data)
        && request.resource.data.diff(resource.data).affectedKeys().hasOnly(['signedBy', 'signedAt'])
        && request.resource.data.signedBy == me().name
        && request.resource.data.signedAt == request.time;
      allow delete: if isManager();
    }

    // Closing photos: a compressed JPEG each, attached to a shift report when it is sent.
    match /photos/{id} {
      allow read: if isMember();
      allow create: if isMember()
        && request.resource.data.keys().hasOnly(['data', 'note', 'report', 'uid', 'name', 'at'])
        && request.resource.data.data is string && request.resource.data.data.size() < 1000000
        && request.resource.data.data.matches('data:image/jpeg;base64,.*')
        && request.resource.data.note == '' && request.resource.data.report == null
        && request.resource.data.uid == request.auth.uid
        && request.resource.data.name == me().name
        && request.resource.data.at == request.time;
      // The person who took it can edit the note, or attach it to their report, until the report is sent.
      allow update: if isMember() && resource.data.uid == request.auth.uid && resource.data.report == null
        && ((request.resource.data.diff(resource.data).affectedKeys().hasOnly(['note']) && request.resource.data.note is string && request.resource.data.note.size() <= 300)
          || (request.resource.data.diff(resource.data).affectedKeys().hasOnly(['report']) && request.resource.data.report is string
            && getAfter(/databases/$(database)/documents/reports/$(request.resource.data.report)).data.uid == request.auth.uid));
      allow delete: if isManager() || (isMember() && resource.data.uid == request.auth.uid && resource.data.report == null);
    }

    // Tonight's shared checklist ticks, one document per business day.
    match /shift/{day} {
      allow read, delete: if isMember();
      allow create, update: if isMember() && request.resource.data.keys().hasOnly(['open', 'close'])
        && request.resource.data.get('open', []) is list && request.resource.data.get('open', []).size() <= 60
        && request.resource.data.get('close', []) is list && request.resource.data.get('close', []).size() <= 60;
    }

    // Stock requests: anyone on the team asks, managers mark them ordered.
    match /requests/{id} {
      allow read: if isMember();
      allow create: if isMember()
        && request.resource.data.keys().hasOnly(['item', 'note', 'done', 'uid', 'name', 'at'])
        && text(request.resource.data.item, 200)
        && request.resource.data.note is string && request.resource.data.note.size() <= 200
        && request.resource.data.done == false
        && request.resource.data.uid == request.auth.uid
        && request.resource.data.name == me().name
        && request.resource.data.at == request.time;
      allow update: if isManager() && resource.data.done == false
        && request.resource.data.diff(resource.data).affectedKeys().hasOnly(['done', 'doneBy', 'doneAt'])
        && request.resource.data.done == true
        && request.resource.data.doneBy == me().name
        && request.resource.data.doneAt == request.time;
      allow delete: if isManager() || (isMember() && resource.data.uid == request.auth.uid && resource.data.done == false);
    }

    // Append-only record of every 86 and back-in-stock. Managers read it.
    match /outlog/{id} {
      allow read: if isManager();
      allow create: if isMember()
        && request.resource.data.keys().hasOnly(['name', 'action', 'by', 'at'])
        && request.resource.data.action in ['out', 'back']
        && text(request.resource.data.name, 80)
        && request.resource.data.by == me().name
        && request.resource.data.at == request.time;
    }

    match /out/{id} {
      allow read, delete: if isMember();
      allow create, update: if isMember()
        && request.resource.data.keys().hasOnly(['name', 'by', 'at'])
        && text(request.resource.data.name, 80)
        && request.resource.data.by == me().name
        && request.resource.data.at == request.time;
    }
  }
}
